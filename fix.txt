   LARABLOG - category.js

'use strict';

(function () {
    // ==================== SORT POSTS (Simulated) ====================
    const sortSelect = document.getElementById('sortPosts');
    const postsGrid = document.getElementById('postsGrid');

    if (sortSelect && postsGrid) {
        sortSelect.addEventListener('change', function () {
            const sortBy = this.value;
            const posts = Array.from(postsGrid.querySelectorAll('.col-md-6'));

            // Simple sorting simulation based on data attribute or random
            // Since we don't have real data, we'll just flip order or shuffle
            if (sortBy === 'oldest') {
                posts.reverse();
            } else if (sortBy === 'popular') {
                // Shuffle for demonstration
                posts.sort(() => Math.random() - 0.5);
            } else {
                // Newest – reset to original order (just reload page? we'll keep as is)
                // For demo, we'll just append them in the order they are now
                // Since they are already in HTML order, we do nothing
                return; // Keep original
            }

            // Clear and re-append
            postsGrid.innerHTML = '';
            posts.forEach(post => postsGrid.appendChild(post));
        });
    }

    // ==================== PAGINATION ACTIVE STATE ====================
    const paginationLinks = document.querySelectorAll('.pagination .page-link');
    paginationLinks.forEach(link => {
        link.addEventListener('click', function (e) {
            // Prevent actual navigation for demo
            e.preventDefault();
            const parent = this.parentElement;
            if (parent.classList.contains('disabled') || parent.classList.contains('active')) return;

            // Remove active from all
            document.querySelectorAll('.pagination .page-item').forEach(item => item.classList.remove('active'));
            // Add active to clicked
            parent.classList.add('active');
        });
    });

})();

'use strict';

(function () {
    const tagFilterInput = document.getElementById('tagFilter');
    const tagCards = document.querySelectorAll('.tag-card-col');
    const noTagsFound = document.getElementById('noTagsFound');

    if (tagFilterInput && tagCards.length > 0) {
        tagFilterInput.addEventListener('input', function () {
            const filterTerm = this.value.trim().toLowerCase();
            let visibleCount = 0;

            tagCards.forEach(card => {
                const tagValue = card.getAttribute('data-tag');
                if (tagValue && tagValue.includes(filterTerm)) {
                    card.style.display = '';
                    card.style.opacity = '1';
                    visibleCount++;
                } else {
                    card.style.display = 'none';
                    card.style.opacity = '0';
                }
            });

            // Show/hide no results message
            if (noTagsFound) {
                if (visibleCount === 0) {
                    noTagsFound.style.display = 'block';
                } else {
                    noTagsFound.style.display = 'none';
                }
            }
        });
    }
})();
